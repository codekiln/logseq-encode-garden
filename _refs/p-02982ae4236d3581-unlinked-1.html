			  alias       = "mcp"
			  run = '''
			  #!/usr/bin/env bash
			  set -euo pipefail
			  op run --env-file=github-mcp.env -- \
			    docker run -i --rm \
			      -e GITHUB_PERSONAL_ACCESS_TOKEN \
			      ghcr.io/github/github-mcp-server
			  '''
